Okay, machen wir einen wirklich minimalen, eigenständigen C++‑Mining‑Kernel:
Input: fertiger 80‑Byte‑Blockheader + Target (als 256‑Bit Zahl) → Output: gefundene Nonce (falls eine).

Ich halte alles so flach wie möglich, ohne Abstraktionen.
1. SHA‑256 (doppelt) – minimaler Kern

Das ist ein kompakter, aber vollständiger SHA‑256‑Kern (nur für kurze Inputs, wie unseren 80‑Byte‑Header):
cpp

static inline uint32_t rotr(uint32_t x, uint32_t n) {
    return (x >> n) | (x << (32u - n));
}

static void sha256_block(const uint8_t *data, uint32_t H[8]) {
    uint32_t w[64];

    for (int i = 0; i < 16; ++i) {
        w[i] = (uint32_t(data[4*i]) << 24) |
               (uint32_t(data[4*i+1]) << 16) |
               (uint32_t(data[4*i+2]) << 8) |
               (uint32_t(data[4*i+3]));
    }
    for (int i = 16; i < 64; ++i) {
        uint32_t s0 = rotr(w[i-15], 7) ^ rotr(w[i-15], 18) ^ (w[i-15] >> 3);
        uint32_t s1 = rotr(w[i-2], 17) ^ rotr(w[i-2], 19) ^ (w[i-2] >> 10);
        w[i] = w[i-16] + s0 + w[i-7] + s1;
    }

    uint32_t a = H[0], b = H[1], c = H[2], d = H[3];
    uint32_t e = H[4], f = H[5], g = H[6], h = H[7];

    for (int i = 0; i < 64; ++i) {
        uint32_t S1 = rotr(e, 6) ^ rotr(e, 11) ^ rotr(e, 25);
        uint32_t ch = (e & f) ^ ((~e) & g);
        uint32_t temp1 = h + S1 + ch + K[i] + w[i];
        uint32_t S0 = rotr(a, 2) ^ rotr(a, 13) ^ rotr(a, 22);
        uint32_t maj = (a & b) ^ (a & c) ^ (b & c);
        uint32_t temp2 = S0 + maj;

        h = g;
        g = f;
        f = e;
        e = d + temp1;
        d = c;
        c = b;
        b = a;
        a = temp1 + temp2;
    }

    H[0] += a; H[1] += b; H[2] += c; H[3] += d;
    H[4] += e; H[5] += f; H[6] += g; H[7] += h;
}

static void sha256(const uint8_t *data, size_t len, uint8_t out[32]) {
    uint32_t H[8] = {
        0x6a09e667u,0xbb67ae85u,0x3c6ef372u,0xa54ff53au,
        0x510e527fu,0x9b05688cu,0x1f83d9abu,0x5be0cd19u
    };

    uint8_t block[64];
    size_t full = len / 64;
    for (size_t i = 0; i < full; ++i) {
        sha256_block(data + 64*i, H);
    }

    size_t rem = len % 64;
    std::memset(block, 0, 64);
    if (rem) std::memcpy(block, data + 64*full, rem);
    block[rem] = 0x80;

    uint64_t bitlen = uint64_t(len) * 8u;
    if (rem >= 56) {
        sha256_block(block, H);
        std::memset(block, 0, 64);
    }
    block[56] = uint8_t(bitlen >> 56);
    block[57] = uint8_t(bitlen >> 48);
    block[58] = uint8_t(bitlen >> 40);
    block[59] = uint8_t(bitlen >> 32);
    block[60] = uint8_t(bitlen >> 24);
    block[61] = uint8_t(bitlen >> 16);
    block[62] = uint8_t(bitlen >> 8);
    block[63] = uint8_t(bitlen);

    sha256_block(block, H);

    for (int i = 0; i < 8; ++i) {
        out[4*i+0] = uint8_t(H[i] >> 24);
        out[4*i+1] = uint8_t(H[i] >> 16);
        out[4*i+2] = uint8_t(H[i] >> 8);
        out[4*i+3] = uint8_t(H[i]);
    }
}

static void sha256d(const uint8_t *data, size_t len, uint8_t out[32]) {
    uint8_t tmp[32];
    sha256(data, len, tmp);
    sha256(tmp, 32, out);
}

2. 256‑Bit Target und Vergleich

Bitcoin vergleicht hash <= target (beide als 256‑Bit big‑endian). Wir nehmen einfach 32‑Byte‑Arrays:
cpp

static int cmp256(const uint8_t a[32], const uint8_t b[32]) {
    for (int i = 0; i < 32; ++i) {
        if (a[i] < b[i]) return -1;
        if (a[i] > b[i]) return 1;
    }
    return 0;
}

3. Minimaler Mining‑Loop

Wir erwarten:

    header[80]: kompletter Blockheader in little‑endian, wie Bitcoin ihn intern benutzt
    (Version, PrevHash, MerkleRoot, Time, Bits, Nonce).

    Wir überschreiben nur die letzten 4 Bytes (Nonce).

    target[32]: 256‑Bit Target in big‑endian.

cpp

#include <optional>

struct MiningResult {
    uint32_t nonce;
    uint8_t hash[32]; // double-SHA256, big-endian
};

std::optional<MiningResult> mine_blockheader(
    uint8_t header[80],
    const uint8_t target[32],
    uint32_t start_nonce,
    uint32_t max_tries
) {
    uint8_t hash[32];

    uint32_t nonce = start_nonce;
    for (uint32_t i = 0; i < max_tries; ++i, ++nonce) {
        // Nonce in little-endian in die letzten 4 Bytes des Headers schreiben
        header[76] = uint8_t(nonce);
        header[77] = uint8_t(nonce >> 8);
        header[78] = uint8_t(nonce >> 16);
        header[79] = uint8_t(nonce >> 24);

        sha256d(header, 80, hash);

        if (cmp256(hash, target) <= 0) {
            MiningResult res;
            res.nonce = nonce;
            std::memcpy(res.hash, hash, 32);
            return res;
        }
    }
    return std::nullopt;
}

4. Verwendung (Pseudo‑Beispiel)
cpp

int main() {
    uint8_t header[80];   // hier: korrekt gefüllter Blockheader (little-endian)
    uint8_t target[32];   // aus "bits" dekodiert, big-endian

    // TODO: header & target setzen (z.B. aus Stratum-Job)
