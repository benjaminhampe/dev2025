
#include <iostream>
#include <vector>
#include <immintrin.h> // For AVX and SSE intrinsics

struct DataPoint {
    int64_t nanos;
    double value;
};

struct ScreenPos {
    int32_t x;
    int32_t y;
};

class Transformer {
public:
    Transformer(int64_t minNanos, int64_t maxNanos, double minValue, double maxValue, int32_t screenWidth, int32_t screenHeight)
        : minNanos(minNanos), maxNanos(maxNanos), minValue(minValue), maxValue(maxValue),
          screenWidth(screenWidth), screenHeight(screenHeight) {
        detectInstructionSet();
    }

    void transform(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
        if (useAVX2) {
            transformAVX2(data, screenPos);
        } else if (useSSE) {
            transformSSE(data, screenPos);
        } else {
            transformScalar(data, screenPos);
        }
    }

    void inverseTransform(const std::vector<ScreenPos> &screenPos, std::vector<DataPoint> &data) const {
        if (useAVX2) {
            inverseTransformAVX2(screenPos, data);
        } else if (useSSE) {
            inverseTransformSSE(screenPos, data);
        } else {
            inverseTransformScalar(screenPos, data);
        }
    }

private:
    int64_t minNanos, maxNanos;
    double minValue, maxValue;
    int32_t screenWidth, screenHeight;
    bool useAVX2 = false;
    bool useSSE = false;

    void detectInstructionSet() {
        int32_t info[4];
        __cpuid(info, 0);
        int32_t nIds = info[0];

        if (nIds >= 1) {
            __cpuid(info, 1);
            useSSE = (info[3] & ((int)1 << 25)) != 0; // Check SSE
            useAVX2 = (info[2] & ((int)1 << 28)) != 0; // Check AVX

            if (useAVX2) {
                __cpuid_count(info, 7, 0);
                useAVX2 = (info[1] & ((int)1 << 5)) != 0; // Check AVX2
            }
        }
    }

    void transformScalar(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
        for (size_t i = 0; i < data.size(); ++i) {
            screenPos[i].x = static_cast<int32_t>((data[i].nanos - minNanos) * screenWidth / (maxNanos - minNanos));
            screenPos[i].y = screenHeight - static_cast<int32_t>((data[i].value - minValue) * screenHeight / (maxValue - minValue));
        }
    }

    void inverseTransformScalar(const std::vector<ScreenPos> &screenPos, std::vector<DataPoint> &data) const {
        for (size_t i = 0; i < screenPos.size(); ++i) {
            data[i].nanos = static_cast<int64_t>((screenPos[i].x * (maxNanos - minNanos) / screenWidth) + minNanos);
            data[i].value = static_cast<double>(screenHeight - screenPos[i].y) * (maxValue - minValue) / screenHeight + minValue;
        }
    }

    void transformSSE(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
#ifdef __SSE2__
        const __m128d scaleX = _mm_set1_pd(static_cast<double>(screenWidth) / (maxNanos - minNanos));
        const __m128d scaleY = _mm_set1_pd(static_cast<double>(screenHeight) / (maxValue - minValue));
        const __m128d minNanosVec = _mm_set1_pd(static_cast<double>(minNanos));
        const __m128d minValueVec = _mm_set1_pd(minValue);
        const __m128i screenHeightVec = _mm_set1_epi32(screenHeight);

        for (size_t i = 0; i < data.size(); i += 2) {
            __m128d nanos = _mm_set_pd(static_cast<double>(data[i + 1].nanos), static_cast<double>(data[i].nanos));
            __m128d values = _mm_set_pd(data[i + 1].value, data[i].value);

            __m128d x = _mm_mul_pd(_mm_sub_pd(nanos, minNanosVec), scaleX);
            __m128d y = _mm_mul_pd(_mm_sub_pd(values, minValueVec), scaleY);
            y = _mm_sub_pd(_mm_set1_pd(screenHeight), y);

            __m128i xInt = _mm_cvttpd_epi32(x);
            __m128i yInt = _mm_cvttpd_epi32(y);

            _mm_storeu_si128(reinterpret_cast<__m128i*>(&screenPos[i]), _mm_unpacklo_epi32(xInt, yInt));
        }
#endif
    }

    void inverseTransformSSE(const std::vector<ScreenPos> &screenPos, std::vector<DataPoint> &data) const {
#ifdef __SSE2__
        const __m128d invScaleX = _mm_set1_pd(static_cast<double>(maxNanos - minNanos) / screenWidth);
        const __m128d invScaleY = _mm_set1_pd(static_cast<double>(maxValue - minValue) / screenHeight);
        const __m128d minNanosVec = _mm_set1_pd(static_cast<double>(minNanos));
        const __m128d minValueVec = _mm_set1_pd(minValue);
        const __m128d screenHeightVec = _mm_set1_pd(screenHeight);

        for (size_t i = 0; i < screenPos.size(); i += 2) {
            __m128i x = _mm_set_epi32(0, screenPos[i + 1].x, 0, screenPos[i].x);
            __m128i y = _mm_set_epi32(0, screenPos[i + 1].y, 0, screenPos[i].y);

            __m128d xDbl = _mm_cvtepi32_pd(x);
            __m128d yDbl = _mm_cvtepi32_pd(y);

            __m128d nanos = _mm_add_pd(_mm_mul_pd(xDbl, invScaleX), minNanosVec);
            __m128d values = _mm_add_pd(_mm_mul_pd(_mm_sub_pd(screenHeightVec, yDbl), invScaleY), minValueVec);

            _mm_storeu_pd(reinterpret_cast<double*>(&data[i].nanos), nanos);
            _mm_storeu_pd(&data[i].value, values);
        }
#endif
    }

    void transformAVX2(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
#ifdef __AVX2__
        const __m256d scaleX = _mm256_set1_pd(static_cast<double>(screenWidth) / (maxNanos - minNanos));
        const __m256d scaleY = _mm256_set1_pd(static_cast<double>(screenHeight) / (maxValue - minValue));
        const __m256d minNanosVec = _mm256_set1_pd(static_cast<double>(minNanos));
        const __m256d minValueVec = _mm256_set1_pd(minValue);
        const __m256d screenHeightVec = _mm256_set1_pd(screenHeight);

        for (size_t i = 0; i < data.size(); i += 4) {
            __m256d nanos = _mm256_set_pd(static_cast<double>(data[i + 3].nanos), static_cast<double>(data[i + 2].nanos), static_cast<double>(data[i + 1].nanos), static_cast<double>(data[i].nanos));
            __m256d values = _mm256_set_pd(data[i + 3].value, data[i + 2].value, data[i + 1].value, data[i].value);

            __m256d x = _mm256_mul_pd(_mm256_sub_pd(nanos, minNanosVec), scaleX);
            __m256d y = _mm256_mul_pd(_mm256_sub_pd(values, minValueVec), scaleY);
            y = _mm256_sub_pd(screenHeightVec, y);

            __m128i xInt = _mm256_cvttpd_epi32(x);
            __m128i yInt = _mm256_cvttpd_epi32(y);

            _mm_storeu_si128(reinterpret_cast<__m128i*>(&screenPos[i]), _mm_unpacklo_epi32(xInt, yInt));
            _mm_storeu_si128(reinterpret_cast<__m128i*>(&screenPos[i + 2]), _mm_unpackhi_epi32(xInt, yInt));
        }
#endif
    }
	
	void inverseTransformAVX2(const std::vector<ScreenPos> &screenPos, std::vector<DataPoint> &data) const {
#ifdef __AVX2__
        const __m256d invScaleX = _mm256_set1_pd(static_cast<double>(maxNanos - minNanos) / screenWidth);
        const __m256d invScaleY = _mm256_set1_pd(static_cast<double>(maxValue - minValue) / screenHeight);
        const __m256d minNanosVec = _mm256_set1_pd(static_cast<double>(minNanos));
        const __m256d minValueVec = _mm256_set1_pd(minValue);
        const __m256d screenHeightVec = _mm256_set1_pd(screenHeight);

        for (size_t i = 0; i < screenPos.size(); i += 4) {
            __m256i x = _mm256_set_epi64x(screenPos[i + 3].x, screenPos[i + 2].x, screenPos[i + 1].x, screenPos[i].x);
            __m256i y = _mm256_set_epi64x(screenPos[i + 3].y, screenPos[i + 2].y, screenPos[i + 1].y, screenPos[i].y);

            __m256d xDbl = _mm256_cvtepi64_pd(x);
            __m256d yDbl = _mm256_cvtepi64_pd(y);

            __m256d nanos = _mm256_add_pd(_mm256_mul_pd(xDbl, invScaleX), minNanosVec);
            __m256d values = _mm256_add_pd(_mm256_mul_pd(_mm256_sub_pd(screenHeightVec, yDbl), invScaleY), minValueVec);

            _mm256_storeu_pd(reinterpret_cast<double*>(&data[i].nanos), nanos);
            _mm256_storeu_pd(&data[i].value, values);
        }
#endif
    }
};

// Main function
int main() {
    std::vector<DataPoint> data = {
        {1672531199000000000, 3.14159},
        {1672531200000000000, 2.71828},
        {1672531201000000000, 1.61803},
        {1672531202000000000, 0.57721},
        // More data points...
    };

    int64_t minNanos = 1672531198000000000;
    int64_t maxNanos = 1672531203000000000;
    double minValue = 0.0;
    double maxValue = 10.0;
    int32_t screenWidth = 800;
    int32_t screenHeight = 600;

    Transformer transformer(minNanos, maxNanos, minValue, maxValue, screenWidth, screenHeight);

    std::vector<ScreenPos> screenPos(data.size());
    transformer.transform(data, screenPos);

    for (const auto &pos : screenPos) {
        std::cout << "Screen Position: (" << pos.x << ", " << pos.y << ")\n";
    }

    // Convert back from screen positions to data points
    std::vector<DataPoint> newData(screenPos.size());
    transformer.inverseTransform(screenPos, newData);

    for (const auto &point : newData) {
        std::cout << "Data Point: (nanos: " << point.nanos << ", value: " << point.value << ")\n";
    }

    return 0;
}
