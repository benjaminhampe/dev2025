#include <iostream>
#include <vector>
#include <immintrin.h> // For AVX and SSE intrinsics

struct DataPoint {
    int64_t nanos;
    double value;
};

struct ScreenPos {
    int32_t x;
    int32_t y;
};

class Transformer {
public:
    Transformer(int64_t minNanos, int64_t maxNanos, double minValue, double maxValue, int32_t screenWidth, int32_t screenHeight)
        : m_minNanos(minNanos), m_maxNanos(maxNanos), m_minValue(minValue), m_maxValue(maxValue),
          m_screenWidth(screenWidth), m_screenHeight(screenHeight) {
        detectInstructionSet();
        precomputeFactors();
    }

    void transform(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
        if (m_useAVX2) {
            transformAVX2(data, screenPos);
        } else if (m_useSSE) {
            transformSSE(data, screenPos);
        } else {
            transformScalar(data, screenPos);
        }
    }

    void inverseTransform(const std::vector<ScreenPos> &screenPos, std::vector<DataPoint> &data) const {
        if (m_useAVX2) {
            inverseTransformAVX2(screenPos, data);
        } else if (m_useSSE) {
            inverseTransformSSE(screenPos, data);
        } else {
            inverseTransformScalar(screenPos, data);
        }
    }

private:
    int64_t m_minNanos, m_maxNanos;
    double m_minValue, m_maxValue;
    int32_t m_screenWidth, m_screenHeight;
    bool m_useAVX2 = false;
    bool m_useSSE = false;
    double m_scaleX, m_scaleY, m_invScaleX, m_invScaleY;

    void detectInstructionSet() {
        int32_t info[4];
        __cpuid(info, 0);
        int32_t nIds = info[0];

        if (nIds >= 1) {
            __cpuid(info, 1);
            m_useSSE = (info[3] & ((int)1 << 25)) != 0; // Check SSE
            m_useAVX2 = (info[2] & ((int)1 << 28)) != 0; // Check AVX

            if (m_useAVX2) {
                __cpuid_count(info, 7, 0);
                m_useAVX2 = (info[1] & ((int)1 << 5)) != 0; // Check AVX2
            }
        }
    }

    void precomputeFactors() {
        m_scaleX = static_cast<double>(m_screenWidth) / (m_maxNanos - m_minNanos);
        m_scaleY = static_cast<double>(m_screenHeight) / (m_maxValue - m_minValue);
        m_invScaleX = static_cast<double>(m_maxNanos - m_minNanos) / m_screenWidth;
        m_invScaleY = static_cast<double>(m_maxValue - m_minValue) / m_screenHeight;
    }

    void transformScalar(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
        for (size_t i = 0; i < data.size(); ++i) {
            screenPos[i].x = static_cast<int32_t>((data[i].nanos - m_minNanos) * m_scaleX);
            screenPos[i].y = m_screenHeight - static_cast<int32_t>((data[i].value - m_minValue) * m_scaleY);
        }
    }

    void inverseTransformScalar(const std::vector<ScreenPos> &screenPos, std::vector<DataPoint> &data) const {
        for (size_t i = 0; i < screenPos.size(); ++i) {
            data[i].nanos = static_cast<int64_t>(screenPos[i].x * m_invScaleX + m_minNanos);
            data[i].value = static_cast<double>(m_screenHeight - screenPos[i].y) * m_invScaleY + m_minValue;
        }
    }

    void transformSSE(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
#ifdef __SSE2__
        const __m128d scaleXVec = _mm_set1_pd(m_scaleX);
        const __m128d scaleYVec = _mm_set1_pd(m_scaleY);
        const __m128d minNanosVec = _mm_set1_pd(static_cast<double>(m_minNanos));
        const __m128d minValueVec = _mm_set1_pd(m_minValue);
        const __m128i screenHeightVec = _mm_set1_epi32(m_screenHeight);

        for (size_t i = 0; i < data.size(); i += 2) {
            __m128d nanos = _mm_set_pd(static_cast<double>(data[i + 1].nanos), static_cast<double>(data[i].nanos));
            __m128d values = _mm_set_pd(data[i + 1].value, data[i].value);

            __m128d x = _mm_mul_pd(_mm_sub_pd(nanos, minNanosVec), scaleXVec);
            __m128d y = _mm_mul_pd(_mm_sub_pd(values, minValueVec), scaleYVec);
            y = _mm_sub_pd(_mm_set1_pd(m_screenHeight), y);

            __m128i xInt = _mm_cvttpd_epi32(x);
            __m128i yInt = _mm_cvttpd_epi32(y);

            _mm_storeu_si128(reinterpret_cast<__m128i*>(&screenPos[i]), _mm_unpacklo_epi32(xInt, yInt));
        }
#endif
    }

    void inverseTransformSSE(const std::vector<ScreenPos> &screenPos, std::vector<DataPoint> &data) const {
#ifdef __SSE2__
        const __m128d invScaleXVec = _mm_set1_pd(m_invScaleX);
        const __m128d invScaleYVec = _mm_set1_pd(m_invScaleY);
        const __m128d minNanosVec = _mm_set1_pd(static_cast<double>(m_minNanos));
        const __m128d minValueVec = _mm_set1_pd(m_minValue);
        const __m128d screenHeightVec = _mm_set1_pd(m_screenHeight);

        for (size_t i = 0; i < screenPos.size(); i += 2) {
            __m128i x = _mm_set_epi32(0, screenPos[i + 1].x, 0, screenPos[i].x);
            __m128i y = _mm_set_epi32(0, screenPos[i + 1].y, 0, screenPos[i].y);

            __m128d xDbl = _mm_cvtepi32_pd(x);
            __m128d yDbl = _mm_cvtepi32_pd(y);

            __m128d nanos = _mm_add_pd(_mm_mul_pd(xDbl, invScaleXVec), minNanosVec);
            __m128d values = _mm_add_pd(_mm_mul_pd(_mm_sub_pd(screenHeightVec, yDbl), invScaleYVec), minValueVec);

            _mm_storeu_pd(reinterpret_cast<double*>(&data[i].nanos), nanos);
            _mm_storeu_pd(&data[i].value, values);
        }
#endif
    }

    void transformAVX2(const std::vector<DataPoint> &data, std::vector<ScreenPos> &screenPos) const {
#ifdef __AVX2__
        const __m256d scaleXVec = _mm256_set1_pd(m_scaleX);
        const __m256d scaleYVec = _mm256_set1_pd(m_scaleY);

        const __m256d minValueVec = _mm256_set1_pd(m_minValue);

            __m256d x = _mm256_mul_pd(_mm256_sub_pd(nanos, minNanosVec), scaleXVec);
            __m256d y = _mm
